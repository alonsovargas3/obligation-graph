2. **Green.** The coordinator reruns the Task's acceptance commands itself in the review checkout (`uv sync --locked`, then the Task commands). It records collected test IDs, not just counts.
3. **Mid-flight checks** every ~10 minutes and at each heartbeat gap:
   - `worker-read --dispatch <id> --limit 200`.

4. **Two-stage review before merge.**
   - (a) Spec compliance: the diff does exactly what the Task says and matches its Interfaces block.
   - (b) Quality: `/code-review high` on the branch diff, with CLAUDE.md's invariant as the lens.
   - Findings go back to the same worker as a follow-up. The coordinator does not patch worker code.
5. **Model fallback (GLM to Grok).**
   - A quota or credit error is not proof of exit. The coordinator gets positive exit evidence (`worker-show` / `worker-list` liveness `exited`, or the worker's own settlement) and follows the runtime's stop/abandon procedure.
   - Only then does it start a fresh dispatch on Pi `xai/grok-4.7` with `--retry-of`.
   - Before each dispatch the coordinator records the Task's **accepted base SHA**, prerequisite merges included (Tasks 3 and 4 start from the Task 1 merge). It is recorded in `docs/reviews/dispatch-log.md`.
   - The retry starts from the worker's last pushed commit, or from that accepted base SHA if nothing was pushed, never unconditionally from bootstrap.
   - The retry payload names the base SHA, the prerequisite SHAs, and the freeze manifest check. The retry verifies its branch against them before looking at any salvage.
   - Uncommitted work from the failed worker is saved as a patch in `docs/reviews/salvage/<task>.patch` for the retry to inspect, not applied blindly.
   - Stale messages from the old dispatch are rejected.
6. **Report per Task:** outcome; base, red, and final SHAs; collected test IDs and counts; review findings and how each was resolved; the model that produced each merged commit.

### Task 1: TextDoc and schema (`wave1/t1-core`)

**Files:** Create `src/og/textdoc.py`, `src/og/store/schema.sql`, `src/og/store/db.py`. May add: nothing.

**Acceptance tests:** `tests/test_textdoc.py`, `tests/test_schema.py`.

**Interfaces (produces):**
- `Section(id: str, number: str | None, heading: str, char_start: int, char_end: int, article: str | None = None)`, frozen dataclass.
- `Page(page: int, char_start: int, char_end: int)`, `Segment(id: str, section_id: str | None, page: int, char_start: int, char_end: int)`, frozen dataclasses.
- `TextDoc(doc_id, source_sha256, text, sections: list[Section], pages: list[Page], segments: list[Segment])` (mutable dataclass, list defaults via `field(default_factory=list)`), with:
  - `section_for(offset) -> Section | None` (half-open intervals)
  - `section_by_id(id) -> Section | None`
  - `page_for(offset) -> int | None`
  - `segment(id) -> Segment` (raises `KeyError`)
  - `segment_text(id) -> str`
  - `validate() -> None` (raises `ValueError`)
  - `to_json() -> str` (`json.dumps(asdict(self), ensure_ascii=False, indent=1)`)
  - `from_json(s)` and `load(path)` (both validate), `save(path)` (validates first)
- `og.store.db.connect(path="data/graph.db") -> sqlite3.Connection`: autocommit (`isolation_level=None`), `PRAGMA foreign_keys=ON`, schema applied idempotently via `executescript`.

**`validate()` rules:**
- **Empty text:** if `text == ""`, then sections, pages, and segments must all be empty.
- **Pages:** non-empty when text is non-empty, unique page numbers, every page non-empty (`char_start < char_end`), sorted by start. Together they cover the text exactly: the first starts at 0, each starts where the previous ends, the last ends at `len(text)`.
- **Sections:** either empty, or the same coverage rules as pages, with unique ids.
- **Segments:** unique ids, `0 <= char_start < char_end <= len(text)`, and no overlaps when sorted by start.
  - If sections exist, `section_id == section_for(char_start).id` and `char_end <= that section's char_end`. If there are no sections, `section_id is None`.
  - `page == page_for(char_start)` and `char_end <=` that page's end.
- An empty segment list is valid.

```sql
CREATE TABLE IF NOT EXISTS source (

  exhibit TEXT, local_path TEXT NOT NULL, sha256 TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS party (id INTEGER PRIMARY KEY, name TEXT NOT NULL UNIQUE);
CREATE TABLE IF NOT EXISTS agreement (

  effective_date TEXT CHECK (effective_date IS NULL OR date(effective_date) IS effective_date),

CREATE TABLE IF NOT EXISTS clause_ref (

DROP VIEW IF EXISTS visible_guarantees;
DROP VIEW IF EXISTS visible_supersedes;
DROP VIEW IF EXISTS visible_obligation;

CREATE VIEW visible_obligation AS
WITH anchored AS (
  SELECT o.id AS obligation_id,
    CASE WHEN e.date IS NOT NULL
      AND EXISTS (SELECT 1 FROM clause_ref c
                  WHERE c.id = e.clause_ref_id AND c.grounded = 1 AND c.agreement_id = e.agreement_id)
      AND (e.agreement_id = o.agreement_id
           OR e.agreement_id = (SELECT a.base_agreement_id FROM agreement a WHERE a.id = o.agreement_id))
    THEN date(e.date, printf('%+d days', o.offset_days)) END AS anchored_due
  FROM obligation o LEFT JOIN event e ON e.id = o.anchor_event_id
)
SELECT o.*,
  COALESCE(o.due_date, an.anchored_due) AS effective_due,
  CASE WHEN o.status = 'superseded' THEN 'superseded'
       WHEN COALESCE(o.due_date, an.anchored_due) IS NULL THEN 'pending'
       ELSE 'scheduled' END AS lifecycle
FROM obligation o JOIN anchored an ON an.obligation_id = o.id
WHERE EXISTS (SELECT 1 FROM clause_ref c
              WHERE c.obligation_id = o.id AND c.grounded = 1 AND c.agreement_id = o.agreement_id);

CREATE VIEW visible_supersedes AS
SELECT s.* FROM supersedes s JOIN clause_ref c ON c.id = s.clause_ref_id AND c.grounded = 1
WHERE c.agreement_id = s.change_order_id
  AND s.obligation_id IN (SELECT id FROM visible_obligation)
  AND s.superseded_obligation_id IN (SELECT id FROM visible_obligation);

CREATE VIEW visible_guarantees AS
SELECT g.* FROM guarantees g JOIN clause_ref c ON c.id = g.clause_ref_id AND c.grounded = 1
WHERE c.agreement_id = (SELECT agreement_id FROM obligation WHERE id = g.guarantee_obligation_id)
  AND g.guarantee_obligation_id IN (SELECT id FROM visible_obligation)
  AND g.guaranteed_obligation_id IN (SELECT id FROM visible_obligation);

CREATE VIEW visible_triggers AS
SELECT t.* FROM triggers t JOIN clause_ref c ON c.id = t.clause_ref_id AND c.grounded = 1
WHERE c.agreement_id IN (SELECT agreement_id FROM obligation
                        WHERE id IN (t.obligation_id, t.triggered_by_obligation_id))
  AND t.obligation_id IN (SELECT id FROM visible_obligation)
  AND t.triggered_by_obligation_id IN (SELECT id FROM visible_obligation);

CREATE VIEW visible_defined_term AS
SELECT d.* FROM defined_term d
JOIN clause_ref c ON c.id = d.clause_ref_id AND c.grounded = 1 AND c.agreement_id = d.agreement_id;
```

**Interfaces (produces):**
- `RateLimiter(rate_per_s: float, clock=time.monotonic, sleep=time.sleep)`: `ValueError` unless `0 < rate <= 10`. `wait()` sleeps so consecutive calls start at least `1/rate` apart; the first call does not sleep.
- `user_agent() -> str`: reads `OG_EDGAR_USER_AGENT`. `SystemExit` naming the variable if unset, and a `SystemExit` mentioning "email" if no email is present. Does not load `.env`; only `main()` calls `load_dotenv()`.
- `parse_filing_date(index_html) -> str | None`: the `div.info` following the `div.infoHead` whose text is `Filing Date`.
- `resolve_exhibit(index_html, index_url, exhibit) -> str`:
  - Rows (`tr`) whose Type cell text, stripped and uppercased, equals `EX-<exhibit>` exactly. No prefix match: `EX-10.10` is not `EX-10.1`.
  - Zero rows gives `SystemExit("... not found ...")`. More than one gives `SystemExit("... ambiguous ...")`.
  - The link comes from the matched row's `a[href]`, resolved with `urljoin(index_url, href)`. The result must be `https` on host `www.sec.gov`, else `SystemExit` mentioning `sec.gov`.
- `fetch_all(sources_path="data/sources.yaml", *, transport=None, limiter=None) -> list[dict]`:
  - Builds its own `httpx.Client(transport=transport, follow_redirects=True, timeout=30, headers={"User-Agent": ua}, event_hooks={"request": [lambda r: limiter.wait()]})` and closes it. Every request, redirects included, passes the limiter. The default limiter is `RateLimiter(5)`.
  - For each document:
    - If `url` is null and `filing_index_url` is set, GET the index, then set `url` (resolve_exhibit) and `filing_date` (parse_filing_date).
    - If `local_path` exists and its sha256 equals a non-null pin, skip.

**Emitter algorithm (`html.py`).** Positions are final when emitted. Never re-search the text for offsets.
1. Parse with `BeautifulSoup(html, "lxml")`. Skip `head`, `script`, `style`, `title`, `noscript`, and comments.
2. Walk the DOM once, in document order. Keep `lines: list[str]` and a `current` line buffer.
   - **Text nodes:** replace every run of `[ \t\r\n\f\v]` with one space, keeping U+00A0 and every other character as is. Append to `current`.
   - **Block elements** `p, div, li, tr, h1..h6, table, blockquote, center`: a line boundary before and after the element.
   - **`br`:** a line boundary. **`hr`:** a line boundary, then `page_pending = True`.
   - **Page-break styles on any element, inline included (e.g. `span`):**
     - Before-styles: line boundary first (flushing text buffered before the element), then `page_pending = True`, then the element's content.
     - After-styles: the element's content, then a line boundary (flushing it), then `page_pending = True`.
   - **Cells:** within a `tr`, between consecutive `td`/`th` cells that both contain text, emit ` | `.
3. **Line boundary:** strip `current` at both ends of every `str.isspace()` character (NBSP included). If the result is non-empty, append it to `lines`. Clear `current`. Empty and whitespace-only lines are never emitted.
4. **Final text:** `"\n".join(lines)` with no trailing newline. Line start offsets come from the cumulative lengths.
5. **Page breaks** (CSS checks lowercase the `style` value and remove all whitespace first):
   - Before-styles are `page-break-before:always` and `break-before:page`. After-styles are `page-break-after:always` and `break-after:page`. Flush ordering is as in step 2, so text never crosses a page marker.
   - When the next non-empty line is appended while `page_pending`, a new page starts at that line's start offset, unless the offset is 0. Then `page_pending = False`.
   - A pending break with no following text is dropped, coincident breaks collapse, and empty pages are impossible.
   - Pages are `Page(n, start_n, start_{n+1})`, the last ending at `len(text)`.

**Sections (`sections.py`).** Run on the final lines, flat at the finest numbered level:
- `ARTICLE` line: `^(?:ARTICLE|Article)\s+([IVXLC]+|\d+)\b[.:]?\s*(.*)$`.
  - `number` = group 1; `heading` = group 2 stripped.
  - If the heading is empty and the next line is ≤ 80 chars, has no lowercase letters, **and does not itself match any of the three section patterns**, heading = next line. That line stays inside this section. Otherwise the heading stays `""` and the next line starts its own section.
  - `article` = heading, or `None` when the heading is `""`.
- `Section` line: `^(?:Section|SECTION)\s+(\d+(?:\.\d+)*)\.?\s+(.*)$`.
- Numbered clause: `^(\d+(?:\.\d+)+)\.?\s+(\S.*)$`. At least one dot; a bare `1.` is not a section in this wave. If the real corpus needs it, report it; don't add it.
- For Section and numbered lines:
  - heading = group 2 up to the first `". "` or a trailing `"."`, capped at 80 chars, stripped.
  - `article` = the most recent ARTICLE's `article` value (None if there is none, or if that ARTICLE's heading is empty).
- Each section runs from its line's start to the next section's line start, the last to `len(text)`.
- Text before the first heading is `Section("s0000", None, "Preamble", 0, first_start, None)`. It exists only when that span is non-empty. With no headings at all, s0000 covers everything.
- Ids are `s0000` (preamble), then `s0001`, `s0002`, … in order. Without a preamble, numbering still starts at `s0001`.

**Segments:**
- One per emitted line, ids `p0001`… in order.
- Offsets are the line's start and end, newline excluded.
- `section_id` from `section_for(start)`, `page` from `page_for(start)`.

```python
"""Deterministic grounding (ADR-002). Whitespace-normalized exact match only."""

from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Literal

from og.textdoc import TextDoc


@dataclass(frozen=True)
class Span:
    span_text: str
    char_start: int
    char_end: int
    section_id: str | None = None


@dataclass(frozen=True)
class GroundResult:
    grounded: bool
    char_start: int | None
    char_end: int | None
    method: Literal["range", "section", "none"]
    reason: str | None
    section_id: str | None = None
    page: int | None = None


def _norm_map(s: str) -> tuple[str, list[int]]:
    out: list[str] = []
    idx: list[int] = []
    i, n = 0, len(s)
    while i < n:
        if s[i].isspace():
            j = i
            while j < n and s[j].isspace():
                j += 1
            out.append(" ")
            idx.append(i)
            i = j
        else:
            out.append(s[i])
            idx.append(i)
            i += 1
    return "".join(out), idx


def normalize_ws(s: str) -> str:
    return _norm_map(s)[0].strip()


def _find_all(text: str, base: int, needle: str) -> list[tuple[int, int]]:
    hay, idx = _norm_map(text)
    hits, pos = [], hay.find(needle)
    while pos != -1:
        hits.append((base + idx[pos], base + idx[pos + len(needle) - 1] + 1))
        pos = hay.find(needle, pos + 1)
    return hits


def _fail(reason: str) -> GroundResult:
    return GroundResult(False, None, None, "none", reason)


def _ok(doc: TextDoc, start: int, end: int, method: str) -> GroundResult:
    sec = doc.section_for(start)
    return GroundResult(True, start, end, method, None, sec.id if sec else None, doc.page_for(start))


def ground(doc: TextDoc, span: Span) -> GroundResult:
    needle = normalize_ws(span.span_text)
    if not needle:
        return _fail("empty_span")
    explicit = doc.section_by_id(span.section_id) if span.section_id else None
    if 0 <= span.char_start < span.char_end <= len(doc.text):
        window = doc.text[span.char_start : span.char_end]
        for start, end in _find_all(window, span.char_start, needle):
            sec = doc.section_for(start)
            if sec is None or end > sec.char_end:
                continue  # crosses a section boundary
            if explicit is not None and sec.id != explicit.id:
                continue  # an explicit known section constrains the range pass
            return _ok(doc, start, end, "range")
    section = explicit or doc.section_for(span.char_start)  # unknown id: offset's section
    if section is None:
        return _fail("no_section")
    window = doc.text[section.char_start : section.char_end]
    hits = _find_all(window, section.char_start, needle)
    if not hits:
        return _fail("not_found_in_section")
    start, end = min(hits, key=lambda h: (abs(h[0] - span.char_start), h[0]))
    return _ok(doc, start, end, "section")


def log_drop(record: dict, path: str | Path = "logs/dropped.jsonl") -> None:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open("a", encoding="utf-8") as f:
        f.write(json.dumps(record, ensure_ascii=False) + "\n")
```

```python
"""Redaction and form-blank markers. Matching text is reported, never inferred."""

import re

_REDACTED = re.compile(
    r"\[\s*\*(?:\s*\*)*\s*\]|(?<!\*)\*{3,}(?!\*)|\[\s*REDACTED\s*\]"
    r"|\[\s*Confidential Treatment Requested\s*\]",
    re.IGNORECASE,
)
_BLANK = re.compile(r"\[\s*[●•]\s*\]|\[\s+\]|_{4,}")


def is_redacted(s: str) -> bool:
    return bool(_REDACTED.search(s))


def is_blank(s: str) -> bool:
    return bool(_BLANK.search(s))
```
